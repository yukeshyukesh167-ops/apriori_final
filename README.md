## PROGRAM:

from itertools import combinations
# Transactions
T = [
    ['Bread', 'Milk'],
    ['Bread', 'Butter', 'Juice', 'Eggs'],
    ['Milk', 'Butter', 'Juice', 'Jam'],
    ['Bread', 'Milk', 'Butter', 'Juice'],
    ['Bread', 'Milk', 'Butter', 'Jam']
]

# Minimum support count
min_support = 2

# Get all unique items
items = sorted(set(item for transaction in T for item in transaction))

# Apriori algorithm
for k in range(1, len(items) + 1):

    print("\nC" + str(k))

    # Generate candidate itemsets
    candidates = combinations(items, k)

    frequent = []

    # Calculate support
    for c in candidates:
        support = sum(set(c).issubset(transaction) for transaction in T)

        print(c, "=", support)

        # Select frequent itemsets
        if support >= min_support:
            frequent.append((c, support))

    # Display frequent itemsets
    print("L" + str(k))

    for c, support in frequent:
        print(c, "=", support)

    # Stop if no frequent itemsets
    if not frequent:
        break
